// ================================================================ // THE COUSINS TAKE DEEP CREEK — RSVP Google Apps Script // ================================================================ // HOW TO USE: // 1. Go to script.google.com → New Project // 2. Delete the default code and paste ALL of this in // 3. Change SHEET_ID below to your Google Sheet ID // (the long string in the sheet URL between /d/ and /edit) // 4. Click Deploy → New Deployment → Web App // - Execute as: Me // - Who has access: Anyone // 5. Copy the web app URL and paste it into cousins-deep-creek.html // as the GAS_URL value // ================================================================ const SHEET_ID = 'YOUR_GOOGLE_SHEET_ID'; // ← paste your Sheet ID here const SHEET_TAB = 'RSVPs'; function doGet(e) { const action = e.parameter.action; if (action === 'count') { return getCount(); } return jsonResponse({ error: 'Unknown action' }); } function doPost(e) { const action = e.parameter.action; if (action === 'add') { return addRSVP(e); } return jsonResponse({ error: 'Unknown action' }); } function getCount() { try { const sheet = getSheet(); // Count rows with data (minus header row) const lastRow = sheet.getLastRow(); const count = Math.max(0, lastRow - 1); return jsonResponse({ count: count }); } catch(err) { return jsonResponse({ error: err.message }); } } function addRSVP(e) { try { const data = JSON.parse(e.postData.contents); const sheet = getSheet(); // Add header row if sheet is empty if (sheet.getLastRow() === 0) { sheet.appendRow(['Timestamp', 'Name', 'Contact', 'Dietary/Needs', 'Ride', 'Submitted At']); // Format header const header = sheet.getRange(1, 1, 1, 6); header.setFontWeight('bold'); header.setBackground('#1a3a2a'); header.setFontColor('#ffffff'); } sheet.appendRow([ new Date(), data.name || '', data.contact || '', data.diet || '', data.ride || '', data.timestamp || '' ]); const count = Math.max(0, sheet.getLastRow() - 1); return jsonResponse({ success: true, count: count }); } catch(err) { return jsonResponse({ error: err.message }); } } function getSheet() { const ss = SpreadsheetApp.openById(SHEET_ID); let sheet = ss.getSheetByName(SHEET_TAB); if (!sheet) { sheet = ss.insertSheet(SHEET_TAB); } return sheet; } function jsonResponse(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
